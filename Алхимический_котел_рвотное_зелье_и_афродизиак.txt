var api = Java.type('noppes.npcs.api.NpcAPI').Instance();
var playerObject = undefined;

if (typeof RP_registerRecipes !== 'function') {
  var RP_registerRecipes = function (ctx) {};
}

function init(event) {
  playerObject = new getPlayerObject(event.player);
  playerObject.init(event);
}

function interact(event) {
  if (!event.player) return;
  if (!playerObject) playerObject = new getPlayerObject(event.player);

  playerObject.player = event.player;
  playerObject.world = event.player.world;
  playerObject.loadWorldInfo();
  playerObject.processPendingDrops();
  playerObject.interact(event);
}

function damagedEntity(event) {
  if (!event.player) return;
  if (!playerObject) playerObject = new getPlayerObject(event.player);

  playerObject.player = event.player;
  playerObject.world = event.player.world;
  playerObject.loadWorldInfo();
  playerObject.processPendingDrops();
  playerObject.attack(event);
}

function tick(event) {
  if (!event.player) return;
  if (!playerObject) playerObject = new getPlayerObject(event.player);

  playerObject.player = event.player;
  playerObject.world = event.player.world;
  playerObject.loadWorldInfo();
  playerObject.processPendingDrops();
  playerObject.processSpecialPotionEffects();
}

function broken(event) {
  if (!playerObject) return;
  playerObject.broken(event);
}

function timer(event) {
  if (!event.player) return;
  if (!playerObject) playerObject = new getPlayerObject(event.player);

  playerObject.player = event.player;
  playerObject.world = event.player.world;
  playerObject.loadWorldInfo();
  playerObject.processPendingDrops();
  playerObject.processSpecialPotionEffects();
}

function getPlayerObject(player) {
  this.player = player;
  this.name = player.name;
  this.info = {};
  this.world = player.world;

  var self = this;

  this.MAX_INGREDIENTS = 5;
  this.RECIPE_NOTIFY_RADIUS = 4;
  this.CAULDRON_REQUIRED_SKILL = 'MAGIC';
  this.CAULDRON_REQUIRED_LEVEL = 4;

  // ============================================
  // ДОСТУП К КОТЛУ
  // ============================================

  this.isCauldronBlock = function (block) {
    if (!block) return false;
    return block.name === 'minecraft:cauldron' || block.name === 'minecraft:water_cauldron';
  };

  this.getPlayerMcEntity = function (player) {
    if (!player) return null;

    try {
      if (player.getMCEntity) return player.getMCEntity();
    } catch (errMcEntity) {}

    try {
      if (player.getMCPlayer) return player.getMCPlayer();
    } catch (errMcPlayer) {}

    return null;
  };

  this.getReskillableSkillLevel = function (player, skillName) {
    try {
      var SkillModel = Java.type('net.bandit.reskillable.common.capabilities.SkillModel');
      var Skill = Java.type('net.bandit.reskillable.common.commands.skills.Skill');
      var mcPlayer = this.getPlayerMcEntity(player);
      if (!mcPlayer) return -1;

      var model = SkillModel.get(mcPlayer);
      if (!model) return -1;

      var skill = Skill.valueOf(String(skillName).toUpperCase());
      return model.getSkillLevel(skill);
    } catch (errSkill) {
      return -1;
    }
  };

  this.checkMagicCauldronAccess = function (event) {
    var level = this.getReskillableSkillLevel(this.player, this.CAULDRON_REQUIRED_SKILL);

    if (level >= this.CAULDRON_REQUIRED_LEVEL) return true;

    if (event) event.setCanceled(true);

    if (level < 0) {
      this.player.message('§cКотёл недоступен: не удалось проверить навык MAGIC.');
      this.player.message(
        '§7Проверь, что стоит Reskillable и команда §f/skills get ' +
          this.player.name +
          ' MAGIC §7работает.'
      );
    } else {
      this.player.message('§cКотёл недоступен.');
      this.player.message(
        '§7Нужен навык §dMAGIC §7уровня §f' +
          this.CAULDRON_REQUIRED_LEVEL +
          '§7 или выше. Сейчас: §f' +
          level +
          '§7.'
      );
    }

    return false;
  };

  // ============================================
  // ОБЩИЕ ХЕЛПЕРЫ
  // ============================================

  this.addRecipe = function (recipe) {
    this.passiveRecipes.push(recipe);
  };

  this.getItemName = function (item) {
    if (!item) return '';

    try {
      if (item.getName) return String(item.getName());
    } catch (errGetName) {}

    try {
      return String(item.name);
    } catch (errName) {}

    return '';
  };

  this.makeItemNbt = function (itemId, count) {
    if (!count) count = 1;
    return '{id:"' + itemId + '",Count:' + count + 'b}';
  };

  this.makePotionNbt = function (potionId) {
    return '{id:"minecraft:potion",Count:1b,tag:{Potion:"' + potionId + '"}}';
  };

  this.makeWeakPotionNbt = function (potionKey) {
    var preset = this.WEAK_POTION_PRESETS[potionKey];
    if (!preset) return this.makePotionNbt('minecraft:water');

    var parts = [];

    for (var i = 0; i < preset.effects.length; i++) {
      var effect = preset.effects[i];
      parts.push(
        '{Id:' +
          effect.id +
          'b,Amplifier:' +
          effect.amplifier +
          'b,Duration:' +
          effect.duration +
          ',ShowParticles:1b,ShowIcon:1b}'
      );
    }

    var nbt =
      '{id:"minecraft:potion",Count:1b,tag:{Potion:"minecraft:water",CustomPotionEffects:[' +
      parts.join(',') +
      ']';

    if (preset.color) nbt += ',CustomPotionColor:' + preset.color;

    nbt += '}}';
    return nbt;
  };

  this.runWorldCommand = function (world, command) {
    var cmd = command;
    if (cmd.charAt(0) !== '/') cmd = '/' + cmd;

    try {
      api.executeCommandSilent(world, cmd);
      return true;
    } catch (errSilent) {}

    try {
      api.executeCommand(world, cmd);
      return true;
    } catch (errApi) {}

    return false;
  };

  this.getPlayerServer = function (player) {
    if (!player) return null;

    try {
      var mcEntity = player.getMCEntity();
      if (mcEntity && mcEntity.server) return mcEntity.server;
    } catch (errMcEntity) {}

    try {
      var mcLevel = player.world.getMCLevel();
      if (mcLevel && mcLevel.getServer) return mcLevel.getServer();
    } catch (errMcLevel) {}

    return null;
  };

  // ============================================
  // ЗЕЛЬЯ С ЭФФЕКТАМИ BREWIN' AND CHEWIN'
  // ============================================

  this.resolveBnCEffect = function (effectId) {
    try {
      var BnCEffects = Java.type('umpaz.brewinandchewin.common.registry.BnCEffects');
      var id = String(effectId);

      if (id === 'brewinandchewin:raging') return BnCEffects.RAGING.get();
      if (id === 'brewinandchewin:tipsy') return BnCEffects.TIPSY.get();
      if (id === 'brewinandchewin:intoxication') return BnCEffects.INTOXICATION.get();
      if (id === 'brewinandchewin:sweet_heart') return BnCEffects.SWEET_HEART.get();
    } catch (errBnC) {}

    return null;
  };

  this.buildModEffectPotionMcStack = function (effectId, amplifier, durationTicks, color, itemId) {
    try {
      var MobEffectInstance = Java.type('net.minecraft.world.effect.MobEffectInstance');
      var PotionUtils = Java.type('net.minecraft.world.item.alchemy.PotionUtils');
      var Collections = Java.type('java.util.Collections');
      var BuiltInRegistries = Java.type('net.minecraft.core.registries.BuiltInRegistries');
      var ItemStack = Java.type('net.minecraft.world.item.ItemStack');
      var ResourceLocation = Java.type('net.minecraft.resources.ResourceLocation');

      var effect = this.resolveBnCEffect(effectId);
      if (!effect) return null;

      var instance = new MobEffectInstance(effect, durationTicks, amplifier);
      var itemObject = BuiltInRegistries.ITEM.get(new ResourceLocation(String(itemId)));
      if (!itemObject) return null;

      var mcStack = new ItemStack(itemObject, 1);
      PotionUtils.setCustomEffects(mcStack, Collections.singletonList(instance));

      if (color) mcStack.getOrCreateTag().putInt('CustomPotionColor', color);

      return mcStack;
    } catch (errBuildPotion) {
      return null;
    }
  };

  this.applyDisplayNameToMcStack = function (mcStack, displayName, namePrefix) {
    if (!mcStack || !displayName) return;

    try {
      var CompoundTag = Java.type('net.minecraft.nbt.CompoundTag');
      var tag = mcStack.getOrCreateTag();
      var display = new CompoundTag();
      var colorName = 'white';

  this.getEffectListFromEntityNbt = function (nbt) {
    if (!nbt) return null;

    try {
      if (nbt.has('ActiveEffects')) return nbt.getList('ActiveEffects', 10);
    } catch (errActive) {}

    try {
      if (nbt.has('active_effects')) return nbt.getList('active_effects', 10);
    } catch (errActive2) {}

    return null;
  };

  this.readEffectNumericId = function (entry) {
    if (!entry) return -1;

    try {
      return entry.getInt('Id');
    } catch (errInt) {}

    try {
      return entry.getByte('Id');
    } catch (errByte) {}

    return -1;
  };

  this.readEffectAmplifier = function (entry) {
    if (!entry) return -1;

    try {
      return entry.getInt('Amplifier');
    } catch (errInt) {}

    try {
      return entry.getByte('Amplifier');
    } catch (errByte) {}

    try {
      return entry.getInt('amplifier');
    } catch (errLowerInt) {}

    try {
      return entry.getByte('amplifier');
    } catch (errLowerByte) {}

    return -1;
  };

  this.effectEntryMatchesStringId = function (entry, effectId) {
    if (!entry || !effectId) return false;

    try {
      if (String(entry.getString('id')) === String(effectId)) return true;
    } catch (errId) {}

    try {
      if (String(entry.getString('forge:id')) === String(effectId)) return true;
    } catch (errForgeId) {}

    try {
      if (String(entry.toJsonString()).indexOf(String(effectId)) >= 0) return true;
    } catch (errJson) {}

    return false;
  };

  this.hasEffect = function (effectId, numericId, requiredAmplifier) {
    try {
      var list = this.getEffectListFromEntityNbt(this.player.getEntityNbt());
      if (!list) return false;

      for (var i = 0; i < list.size(); i++) {
        var entry = list.get(i);
        var idMatches =
          this.effectEntryMatchesStringId(entry, effectId) ||
          this.readEffectNumericId(entry) === numericId;

        if (!idMatches) continue;

        if (requiredAmplifier === undefined || requiredAmplifier === null) return true;
        if (this.readEffectAmplifier(entry) === requiredAmplifier) return true;
      }
    } catch (errEffects) {}

    return false;
  };

        itemMC instanceof BowItem ||
        itemMC instanceof CrossbowItem ||
        itemMC instanceof TridentItem

      }

    var lowerName = String(itemName).toLowerCase();
    var nameKeys = [
      'sword',
      'blade',
      'dagger',
      'knife',
      'katana',
      'saber',
      'rapier',
      'axe',
      'mace',
      'hammer',

  this.attack = function (event) {
    var target = event.target;
    if (!target) return;

    var item = this.player.getMainhandItem();
    if (!item || item.isEmpty()) return;

    var nbt = item.getNbt();
    var expireMs = 0;
    var hits = 0;
    var duration = 5;
    var level = 0;

    try {
      expireMs = nbt.getLong('RP_PoisonExpireMs');
      hits = nbt.getInteger('RP_PoisonHits');
      duration = nbt.getInteger('RP_PoisonDurationSeconds');
      level = nbt.getInteger('RP_PoisonLevel');
    } catch (errReadPoison) {}

    if (expireMs <= 0 || hits <= 0) return;

    if (new Date().getTime() > expireMs) {
      this.clearWeaponPoison(item);
      this.player.message('§7Яд на оружии выдохся.');
      return;
    }

    try {
      target.addPotionEffect(19, duration, level, true);
    } catch (errPoison) {
      return;
    }

    hits--;

    if (hits <= 0) {
      this.clearWeaponPoison(item);
      this.player.message('§7Яд на оружии закончился.');
    } else {
      nbt.setInteger('RP_PoisonHits', hits);
      this.player.message('§2Цель отравлена. Зарядов яда: ' + hits + '.');
    }
  };

  this.init = function (event) {
    this.loadWorldInfo();

  this.interact = function (event) {
    if (event.type !== 2) return;

    this.loadWorldInfo();

    var target = event.target;
    var item = this.player.getMainhandItem();

    if (this.isCauldronBlock(target) && !this.checkMagicCauldronAccess(event)) return;
    if (this.checkRestartRecipe(target, item, event)) return;
    if (this.checkCooldownInteract(target, item, event)) return;
    if (this.checkCauldronInteract(target, item, event)) return;
    if (this.checkIsPassiveRecipe(target, item, event)) return;
    if (this.interactCauldron(target, item, event)) return;
    if (this.checkWaterInteract(target, item, event)) return;
  };

  this.checkCooldownInteract = function (target, item, event) {
    if (target.name !== 'minecraft:water_cauldron') return false;

    var cauldron = this.getCauldron(target);

    if (cauldron.cd > new Date().getTime()) {
      event.setCanceled(true);
      return true;
    }

    if (cauldron.pendingDrop && new Date().getTime() < cauldron.pendingDrop.until) {
      event.setCanceled(true);
      return true;
    }

    return false;
  };

  this.broken = function (event) {
    var block = event.block;
    var name = 'x' + block.x + 'y' + block.y + 'z' + block.z;

    if (this.info.hasOwnProperty(name)) {
      delete this.info[name];
      this.saveWorldInfo();
    }
  };

  this.checkRestartRecipe = function (target, item, event) {
    if (!this.player.isSneaking() || target.name !== 'minecraft:water_cauldron') return false;

    var cauldron = this.getCauldron(target);

    if (cauldron.ingredients.length || cauldron.pendingDrop) {
      this.resetCauldron(target, '§7Котёл очищен.');
    } else {
      this.player.message('§7В котле нет ингредиентов.');
    }

    event.setCanceled(true);
    return true;
  };

  this.checkWaterInteract = function (target, item, event) {

      target.name === 'minecraft:water_cauldron' &&
      item &&
      !item.isEmpty() &&
      (item.name === 'minecraft:water_bucket' ||
        item.name === 'minecraft:potion' ||
        item.name === 'minecraft:glass_bottle')
    ) {

        this.player.message('Нельзя заливать или вычерпывать котёл, пока внутри есть ингредиенты.');

    }

    return false;
  };

  this.checkIsPassiveRecipe = function (target, item, event) {

      item &&

  this.setCauldronPlayer = function (cauldron, player) {
    if (!cauldron || !player) return;

  this.notifyRecipeReady = function (block, recipeName) {
    var cauldron = this.getCauldron(block);
    var message = '§aОтвар готов: ' + recipeName + '. Заберите результат.';
    var notified = {};

    try {
      var pos = api.getIPos(block.x + 0.5, block.y + 1, block.z + 0.5);
      var nearby = this.world.getNearbyEntities(pos, this.RECIPE_NOTIFY_RADIUS, 1);

      if (nearby && nearby.length) {
        for (var i = 0; i < nearby.length; i++) {
          var player = nearby[i];
          if (!player) continue;

          var playerName = String(player.getName());
          if (notified[playerName]) continue;

          notified[playerName] = true;
          player.message(message);
        }
      }
    } catch (errNearby) {}

    if (cauldron.lastPlayer && !notified[cauldron.lastPlayer]) {
      try {
        var players = this.world.getAllPlayers();

        for (var j = 0; j < players.length; j++) {
          var lastPlayer = players[j];

          if (lastPlayer && String(lastPlayer.getName()) === String(cauldron.lastPlayer)) {
            lastPlayer.message(message);
            break;
          }
        }
      } catch (errLastPlayer) {}
    }
  };

  this.checkPassiveInteract = function (block, event) {
    var cauldron = this.getCauldron(block);
    if (cauldron.ingredients.length === 0) return false;

    var ingredients = cauldron.ingredients;
    this.setCauldronPlayer(cauldron, this.player);

    for (var recipe in this.passiveRecipes) {
      var info = this.passiveRecipes[recipe];

      if (!this.checkItems(info.items, ingredients)) continue;

      if (info.needsBottle && !this.hasGlassBottle(this.player)) {
        this.player.message('§cНужна стеклянная бутылка в руке.');
        event.setCanceled(true);
        return true;
      }

      if (info.func(this.player)) {
        if (info.needsBottle) this.consumeGlassBottle(this.player);

        this.setWaterLevel(block, this.getWaterLevel(block) - info.waterDisc);
        if (this.getWaterLevel(block) === 0) this.clearCauldron(block);
      }

      event.setCanceled(true);
      return true;
    }

    return false;
  };

  this.checkItems = function (targetItems, nowItems) {
    if (targetItems.length !== nowItems.length) return false;

    for (var i = 0; i < targetItems.length; i++) {
      if (targetItems[i] !== nowItems[i]) return false;
    }

    return true;
  };

  this.setWaterLevel = function (block, level) {
    if (level <= 0) {
      this.clear(block);
      return;
    }

    var IntegerClass = Java.type('java.lang.Integer');
    var integerLevel = new IntegerClass(level);
    var blockState = block.getMCBlockState();
    var mcPos = block.getPos().getMCBlockPos();
    var levelProperty = Java.type(
      'net.minecraft.world.level.block.state.properties.BlockStateProperties'
    ).f_61418_;
    var newBlockState = blockState.m_61124_(levelProperty, integerLevel);

    this.player.world.getMCLevel().m_7731_(mcPos, newBlockState, 3);
  };

  this.processPendingDrops = function () {
    var now = new Date().getTime();
    var changed = false;

    for (var key in this.info) {
      if (!this.info.hasOwnProperty(key)) continue;

      var cauldron = this.info[key];
      if (!cauldron || !cauldron.pendingDrop) continue;
      if (now < cauldron.pendingDrop.until) continue;

      var drop = cauldron.pendingDrop;
      cauldron.pendingDrop = null;

      this.despawnDropEntityAt(drop.x, drop.y, drop.z);

      var block = this.player.world.getBlock(drop.x, drop.y, drop.z);

      if (block && block.name === 'minecraft:water_cauldron') {
        try {
          this.player.world.playSoundAt(block.getPos(), 'minecraft:entity.generic.splash', 1, 1);
          this.player.world.spawnParticle(
            'bubble_pop',
            drop.x + 0.5,
            drop.y + 1,
            drop.z + 0.5,
            0.2,
            0,
            0.2,
            0,
            30
          );
        } catch (errEffects) {}

        cauldron.ingredients.push(drop.itemName);
        this.checkRecipes(cauldron, block);
      }

      changed = true;
    }

    if (changed) this.saveWorldInfo();
  };

  this.checkFire = function (block) {
    var fireBlock = this.player.world.getBlock(block.x, block.y - 1, block.z);
    if (!fireBlock) return false;

    var fireBlockName = fireBlock.name || fireBlock.getName();

    if (
      fireBlockName !== 'minecraft:campfire' &&
      fireBlockName !== 'minecraft:soul_campfire'
    ) {
      return false;
    }

    try {
      return String(fireBlock.getMCBlockState()).indexOf('lit=true') !== -1;
    } catch (errFireState) {
      return false;
    }
  };

  this.interactCauldron = function (block, item, event) {
    if (
      block.name !== 'minecraft:water_cauldron' ||
      !item ||
      item.isEmpty() ||
      this.isBlackList(item)
    ) {
      return false;
    }

    event.setCanceled(true);

    if (!this.checkFire(block)) {
      this.player.message('Под котлом должен быть горящий костёр.');
      return true;
    }

    if (this.getWaterLevel(block) !== 3) {
      this.player.message('Котёл должен быть полным.');
      return true;
    }

    var cauldron = this.getCauldron(block);
    this.setCauldronPlayer(cauldron, this.player);

    if (cauldron.ingredients.length < this.MAX_INGREDIENTS) {
      cauldron.cd = new Date().getTime() + 1200;
      this.animate(block, item);
      item.setStackSize(item.getStackSize() - 1);
      this.player.swingMainhand();
    }

    this.saveWorldInfo();
    return true;
  };

  this.isBlackList = function (item) {
    if (!item || item.isEmpty()) return false;
    if (this.isPoisonableWeapon(item)) return true;

    try {
      var itemMC = item.getMCItemStack().m_41720_();
      var ArmorItem = Java.type('net.minecraft.world.item.ArmorItem');
      var ShieldItem = Java.type('net.minecraft.world.item.ShieldItem');

      if (itemMC instanceof ArmorItem || itemMC instanceof ShieldItem) return true;
    } catch (errItemClass) {}

    if (item.name === 'minecraft:glass_bottle') return true;

    return false;
  };

  this.dropEntityKey = function (x, y, z) {
    return 'alch_drop_' + x + '_' + y + '_' + z;
  };

  this.despawnDropEntityAt = function (x, y, z) {
    var key = this.dropEntityKey(x, y, z);
    var entity = this.world.getTempdata().get(key);
    if (!entity) return;

    try {
      entity.despawn();
    } catch (errDespawn) {}

    this.world.getTempdata().remove(key);
  };

  this.animate = function (block, item) {
    var cauldron = this.getCauldron(block);
    this.despawnDropEntityAt(block.x, block.y, block.z);

    var entity = this.spawnItemEntity(this.player.world, item, {
      x: block.x + 0.5,
      y: block.y + 1.45,
      z: block.z + 0.5,
    });

    if (entity) {
      this.world.getTempdata().put(this.dropEntityKey(block.x, block.y, block.z), entity);
    }

    cauldron.pendingDrop = {
      itemName: this.getItemName(item),
      until: new Date().getTime() + 250,
      x: block.x,
      y: block.y,
      z: block.z,
    };
  };

  this.spawnItemEntity = function (world, item, pos) {
    var entity = world.createEntity('minecraft:item');
    var itemCopy = item.copy();
    itemCopy.setStackSize(1);
    entity.setItem(itemCopy);

    var nbt = entity.getEntityNbt();
    nbt.setInteger('PickupDelay', 32767);
    nbt.setByte('NoGravity', 0);
    nbt.setInteger('Age', 5995);
    entity.setEntityNbt(nbt);

    entity.setPosition(pos.x, pos.y, pos.z);
    entity.spawn();
    return entity;
  };

  this.getCauldron = function (block) {
    var name = 'x' + block.x + 'y' + block.y + 'z' + block.z;

    if (!this.info[name]) this.info[name] = new this.BlockInfo();

    return this.info[name];
  };

  this.getWaterLevel = function (block) {
    var blockState = block.getMCBlockState();
    var levelProperty = Java.type(
      'net.minecraft.world.level.block.state.properties.BlockStateProperties'
    ).f_61418_;

    if (blockState.m_61138_(levelProperty)) {
      return blockState.m_61143_(levelProperty);
    }

    return 0;
  };

  this.checkRecipes = function (cauldron, block) {
    var ingredients = cauldron.ingredients;

    for (var recipe in this.passiveRecipes) {
      var info = this.passiveRecipes[recipe];

      if (this.checkItems(info.items, ingredients)) {
        this.notifyRecipeReady(block, info.name);
        return true;
      }
    }

    if (cauldron.ingredients.length >= this.MAX_INGREDIENTS) {
      this.failRecipe(block);
    }
  };

  this.playCauldronSteamReset = function (block) {
    var world = this.player.world;
    var x = block.x + 0.5;
    var y = block.y + 1;
    var z = block.z + 0.5;

    try {
      world.playSoundAt(block.getPos(), 'minecraft:block.fire.extinguish', 1, 0.85);
      world.playSoundAt(block.getPos(), 'minecraft:block.lava.extinguish', 0.7, 1.15);
    } catch (errSound) {}

    try {
      world.spawnParticle('cloud', x, y, z, 0.2, 0.25, 0.2, 0.02, 30);
      world.spawnParticle('large_smoke', x, y, z, 0.15, 0.2, 0.15, 0.01, 20);
      world.spawnParticle('bubble_pop', x, y - 0.25, z, 0.12, 0, 0.12, 0, 12);
    } catch (errParticle) {}
  };

  this.resetCauldron = function (block, message) {
    var cauldron = this.getCauldron(block);

    this.despawnDropEntityAt(block.x, block.y, block.z);
    cauldron.ingredients = [];
    cauldron.pendingDrop = null;
    cauldron.cd = 0;

    this.playCauldronSteamReset(block);
    this.clear(block);
    this.saveWorldInfo();

    if (message) this.player.message(message);
  };

  this.failRecipe = function (block) {
    this.clearCauldron(block);
    this.player.world.explode(block.x + 0.5, block.y + 0.7, block.z + 0.5, 2, false, false);
  };

  this.clearCauldron = function (block) {
    var cauldron = this.getCauldron(block);

    this.despawnDropEntityAt(block.x, block.y, block.z);
    cauldron.ingredients = [];
    cauldron.pendingDrop = null;

    this.saveWorldInfo();
    this.clear(block);
  };

  this.clear = function (block) {
    this.player.world.setBlock(block.x, block.y, block.z, 'minecraft:cauldron', 0);
  };

  this.loadWorldInfo = function () {
    if (!this.world) this.world = this.player.world;

    if (!this.world.getTempdata().has('worldInfo')) {
      var raw = this.world.getStoreddata().get('worldInfo');
      var info = {};

      if (raw && String(raw).length > 0) {
        try {
          info = JSON.parse(raw);
        } catch (errParse) {
          info = {};
        }
      }

      this.world.getTempdata().put('worldInfo', info);
    }

    this.info = this.world.getTempdata().get('worldInfo');
  };

  this.BlockInfo = function () {
    this.ingredients = [];
    this.cd = 0;
    this.pendingDrop = null;
    this.lastPlayer = null;
  };

  this.saveWorldInfo = function () {
    this.world.getStoreddata().put('worldInfo', JSON.stringify(this.info));
  };
}
